class Circle:
    def __init__(self, x, y, radius):
        self.centre = Point(x, y)
        self.radius = radius

    def overlap(self, circB):
        d = ((self.centre.x - circB.centre.x)**2
           + (self.centre.y - circB.centre.y)**2
           ) ** 0.5
        return d < (self.radius + circB.radius)

def rank_station_usage(df, year, quarter):
    new_df = df[
        (df["year"] == year) &
        (df["quarter"] == quarter)
    ]
    usage["departure_count"] = (
        new_df["from_station_name"]
            .value_counts()
    )
    return usage.sort_values(
        by=["combined_departure_return_count"],
        ascending=[False]
    )
